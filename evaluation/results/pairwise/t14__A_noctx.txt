```javascript
class ValidationError extends Error {
  constructor(message) {
    super(message);
    this.name = 'ValidationError';
  }
}

function validateAge(age) {
  if (age < 0) {
    throw new ValidationError('年齢は0未満は許容できません');
  }
}

// 使用例
try {
  validateAge(-5);
} catch (err) {
  if (err instanceof ValidationError) {
    console.log('ValidationError が発生しました:', err.message);
  } else {
    console.log('予期しないエラー:', err.message);
  }
}
```
