```jsx
import React from 'react';

// (1) if 文で早期 return する書き方
function ConditionalRenderIf() {
  const isLoggedIn = true;

  if (!isLoggedIn) {
    return null;
  }

  return (
    <div>
      <p>ようこそ</p>
    </div>
  );
}

// (2) JSX 中で && 演算子を使う書き方
function ConditionalRenderAnd() {
  const isLoggedIn = true;

  return (
    <div>
      {isLoggedIn && <p>ようこそ</p>}
    </div>
  );
}

export { ConditionalRenderIf, ConditionalRenderAnd };
```
